class AppState {
  static String name = '';
  static String gender = '';
  static int periods = 8;
  static int duration = 40;
  static TimeOfDay? firstStart;
  static TimeOfDay? lastEnd;
  static TimeOfDay? intervalStart;
  static TimeOfDay? intervalEnd;
  static List<List<Map<String, String>>> timetable = [];
  static final fontSizeNotifier = ValueNotifier<double>(14.0);
  static final Map<String, Set<int>> reliefMap = {};

  // tasks[dateKey][periodIdx] = [{type, content, name}]
  static final Map<String, Map<int, List<Map<String, String>>>> tasks = {};
  // students[className] = [name, ...]
  static final Map<String, List<String>> students = {};
  // attendance[dateKey][periodIdx][studentName] = isPresent
  static final Map<String, Map<int, Map<String, bool>>> attendance = {};

  static String get title => gender == 'Female'
      ? 'Teacher'
      : gender == 'Male'
      ? 'Sir'
      : '';
  static String get displayName =>
      name.isNotEmpty && title.isNotEmpty ? '$name $title' : name;

  static String dateKey(DateTime d) =>
      '${d.year}-${d.month.toString().padLeft(2, '0')}-${d.day.toString().padLeft(2, '0')}';

  static List<Map<String, String>> getTasksFor(DateTime d, int periodIdx) =>
      tasks[dateKey(d)]?[periodIdx] ?? [];

  static void addTask(DateTime d, int periodIdx, Map<String, String> task) {
    final k = dateKey(d);
    tasks.putIfAbsent(k, () => {});
    tasks[k]!.putIfAbsent(periodIdx, () => []);
    tasks[k]![periodIdx]!.add(task);
  }

  static void removeTask(DateTime d, int periodIdx, int taskIndex) {
    final k = dateKey(d);
    tasks[k]?[periodIdx]?.removeAt(taskIndex);
  }

  static bool getAttendance(DateTime d, int periodIdx, String student) =>
      attendance[dateKey(d)]?[periodIdx]?[student] ?? true;

  static void setAttendance(
    DateTime d,
    int periodIdx,
    String student,
    bool present,
  ) {
    final k = dateKey(d);
    attendance.putIfAbsent(k, () => {});
    attendance[k]!.putIfAbsent(periodIdx, () => {});
    attendance[k]![periodIdx]![student] = present;
  }

  // Collect all unique class names from timetable
  static List<String> get allClasses {
    final Set<String> classes = {};
    for (final day in timetable) {
      for (final p in day) {
        final cls = p['class'] ?? '';
        if (cls.isNotEmpty) classes.add(cls);
      }
    }
    return classes.toList()..sort();
  }
}

class OnboardingFlow extends StatefulWidget {
  const OnboardingFlow({super.key});
  @override
  State<OnboardingFlow> createState() => _OnboardingFlowState();
}

class _OnboardingFlowState extends State<OnboardingFlow> {
  final _pageCtrl = PageController();
  int _page = 0;
  static const int kSteps = 8;

  void _next() {
    if (_page < kSteps - 1) {
      _pageCtrl.nextPage(
        duration: const Duration(milliseconds: 400),
        curve: Curves.easeInOutCubic,
      );
    } else {
      Navigator.of(context).pushReplacement(
        MaterialPageRoute(builder: (_) => const TimetableSetupPage()),
      );
    }
  }

  @override
  void dispose() {
    _pageCtrl.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      resizeToAvoidBottomInset: true,
      body: Container(
        decoration: const BoxDecoration(gradient: kGradient),
        child: SafeArea(
          child: Column(
            children: [
              _ProgressBar(current: _page, total: kSteps),
              Expanded(
                child: PageView(
                  controller: _pageCtrl,
                  physics: const NeverScrollableScrollPhysics(),
                  onPageChanged: (i) => setState(() => _page = i),
                  children: [
                    NamePage(onNext: _next),
                    GenderPage(onNext: _next),
                    PeriodsPage(onNext: _next),
                    DurationPage(onNext: _next),
                    FirstStartPage(onNext: _next),
                    LastEndPage(onNext: _next),
                    IntervalStartPage(onNext: _next),
                    IntervalEndPage(onNext: _next),
                  ],
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }
}

    required this.onSave,
    this.isLast = false,

class _TimePageState extends State<_TimePage> {
  TimeOfDay? _sel;
  @override
  void initState() {
    super.initState();
    _sel = widget.initial;
  }

  Future<void> _pick() async {
    final t = await showTimePicker(
      context: context,
      initialTime: _sel ?? const TimeOfDay(hour: 7, minute: 30),
      builder: (ctx, child) => Theme(
        data: Theme.of(ctx).copyWith(
          colorScheme: const ColorScheme.dark(
            primary: Colors.white,
            onPrimary: kPrimaryBlue,
            surface: kMidBlue,
            onSurface: Colors.white,
          ),
        ),
        child: child!,
      ),
    );

class Period {
  final int number;
  final String subject, className;
  final PeriodType type;
  final TimeOfDay startTime, endTime;
  const Period({
    required this.number,
    required this.subject,
    required this.className,
    required this.type,
    required this.startTime,
    required this.endTime,
  });
  Period copyWith({PeriodType? type}) => Period(
    number: number,
    subject: subject,
    className: className,
    type: type ?? this.type,
    startTime: startTime,
    endTime: endTime,
  );
}

// ═══════════════════════════════════════════════════════════════════════════════
// HELPERS
// ═══════════════════════════════════════════════════════════════════════════════
int _toMin(TimeOfDay t) => t.hour * 60 + t.minute;
String _fmt12Tod(TimeOfDay t) {
  final h = t.hour % 12 == 0 ? 12 : t.hour % 12;
  final m = t.minute.toString().padLeft(2, '0');
  final ampm = t.hour < 12 ? "AM" : "PM";
  return "$h:$m $ampm";
}

String _fmt12Now(DateTime dt) {
  final h = dt.hour % 12 == 0 ? 12 : dt.hour % 12;
  final m = dt.minute.toString().padLeft(2, '0');
  final ampm = dt.hour < 12 ? "AM" : "PM";
  return "$h:$m $ampm";
}

List<Period> buildPeriods(DateTime now) {
  final n = AppState.periods > 0 ? AppState.periods : 8;
  final start = AppState.firstStart ?? const TimeOfDay(hour: 7, minute: 30);
  final dur = AppState.duration > 0 ? AppState.duration : 40;
  final intS = AppState.intervalStart;
  final intE = AppState.intervalEnd;
  final dayIdx = now.weekday - 1 < 5 ? now.weekday - 1 : 0;
  final saved = AppState.timetable.length > dayIdx
      ? AppState.timetable[dayIdx]
      : <Map<String, String>>[];
  final List<Period> result = [];
  int cur = _toMin(start);
  for (int i = 0; i < n; i++) {
    if (intS != null && intE != null && cur == _toMin(intS)) cur = _toMin(intE);
    final ps = TimeOfDay(hour: cur ~/ 60, minute: cur % 60);
    final end = cur + dur;
    final pe = TimeOfDay(hour: end ~/ 60, minute: end % 60);
    final m = i < saved.length ? saved[i] : <String, String>{};
    final sub = m['subject'] ?? '';
    final cls = m['class'] ?? '';
    final tp = m['type'] ?? 'subject';
    // Auto OFF if empty or explicitly off
    PeriodType type = (sub.isEmpty || sub == 'Off Period' || tp == 'off')
        ? PeriodType.off
        : PeriodType.regular;
    result.add(
      Period(
        number: i + 1,
        subject: sub,
        className: cls,
        type: type,
        startTime: ps,
        endTime: pe,
      ),
    );
    cur = end;
  }
  // Apply daily relief
  final dk = AppState.dateKey(now);
  final relieved = AppState.reliefMap[dk] ?? {};
  // Mark current + relief
  final nowMins = now.hour * 60 + now.minute;
  return result.asMap().entries.map((e) {
    final i = e.key;
    final p = e.value;
    if (relieved.contains(i) && p.type == PeriodType.regular) {
      return p.copyWith(type: PeriodType.relief);
    }
    if (p.type == PeriodType.regular &&
        nowMins >= _toMin(p.startTime) &&
        nowMins < _toMin(p.endTime)) {
      return p.copyWith(type: PeriodType.current);
    }
    return p;
  }).toList();
}

int getActiveIdx(List<Period> periods, DateTime now) {
  final nowMins = now.hour * 60 + now.minute;
  for (int i = 0; i < periods.length; i++) {
    if (nowMins >= _toMin(periods[i].startTime) &&
        nowMins < _toMin(periods[i].endTime)) {
      return i;
    }
  }
  return periods.length > 3 ? 3 : 0;
}

class TimetablePage extends StatefulWidget {

  State<TimetablePage> createState() => _TimetablePageState();
}

class _TimetablePageState extends State<TimetablePage> {
  late DateTime _now;
  late Timer _timer;
  int _navIdx = 0;
  @override
  void initState() {
    super.initState();
    _now = DateTime.now();
    _timer = Timer.periodic(
      const Duration(seconds: 30),
      (_) => setState(() => _now = DateTime.now()),
    );
  }

  @override
  void dispose() {
    _timer.cancel();
    super.dispose();
  }

  String get _dayName {
    const d = [
      'Monday',
      'Tuesday',
      'Wednesday',
      'Thursday',
      'Friday',
      'Saturday',
      'Sunday',
    ];
    return d[_now.weekday - 1];
  }

  String get _dateLabel {
    const m = [
      'January',
      'February',
      'March',
      'April',
      'May',
      'June',
      'July',
      'August',
      'September',
      'October',
      'November',
      'December',
    ];
    return '${_now.day} ${m[_now.month - 1]} ${_now.year}';
  }

  void _openDetail(Period period, int periodIdx) {
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: Colors.transparent,
      builder: (_) =>
          PeriodDetailSheet(period: period, periodIdx: periodIdx, date: _now),
    );
  }

  @override
  Widget build(BuildContext context) {
    final periods = buildPeriods(_now);
    final activeIdx = getActiveIdx(periods, _now);
    return ValueListenableBuilder<double>(
      valueListenable: AppState.fontSizeNotifier,
      builder: (context, fs, _) => Scaffold(
        backgroundColor: kBgLight,
        body: SafeArea(
          child: Column(
            children: [
              _buildHeader(fs),
              _buildLegend(),
              const SizedBox(height: 8),
              Expanded(
                child: Padding(
                  padding: const EdgeInsets.symmetric(horizontal: 14),
                  child: Row(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    children: [
                      Expanded(
                        child: LayoutBuilder(
                          builder: (_, c) {
                            final n = periods.length;
                            const gap = 5.0;
                            final cardH = (c.maxHeight - gap * (n - 1)) / n;
                            return Column(
                              children: List.generate(
                                n,
                                (i) => Padding(
                                  padding: EdgeInsets.only(
                                    bottom: i < n - 1 ? gap : 0,
                                  ),
                                  child: SizedBox(
                                    height: cardH,
                                    child: GestureDetector(
                                      onTap: () => _openDetail(periods[i], i),
                                      child: PeriodCard(
                                        period: periods[i],
                                        isActive: i == activeIdx,
                                        fontSize: fs,
                                      ),
                                    ),
                                  ),
                                ),
                              ),
                            );
                          },
                        ),
                      ),
                      const SizedBox(width: 8),
                      SizedBox(
                        width: 60,
                        child: TimetableTimeline(
                          currentTime: _fmt12Now(_now),
                          activePeriodIdx: activeIdx,
                          totalPeriods: periods.length,
                        ),
                      ),
                    ],
                  ),
                ),
              ),
              const SizedBox(height: 4),
              _buildNav(),
            ],
          ),
        ),
      ),
    );

        children: List.generate(items.length, (i) {

                    .push(
                      MaterialPageRoute(builder: (_) => const SchedulePage()),

              } else if (i == 2) {

                    .push(MaterialPageRoute(builder: (_) => const TasksPage()))

                    .then((_) => setState(() {}));

            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                Icon(
                  items[i].$1,
                  color: active ? kPrimaryBlue : kMidText,

                    fontWeight: FontWeight.w700,

                    width: 4,
                    height: 4,
                    decoration: BoxDecoration(
                      color: kPrimaryBlue,
                      shape: BoxShape.circle,
                    ),
                  ),
                ],
              ],
            ),
          );
        }),
      ),
    );
  }

class _TasksTab extends StatelessWidget {
  final Period period;
  final int periodIdx;
  final DateTime date;
  final VoidCallback onChanged;
  const _TasksTab({
    required this.period,
    required this.periodIdx,
    required this.date,
    required this.onChanged,
  });

  IconData _icon(String type) {
    switch (type) {
      case 'video_link':
        return Icons.link_rounded;
      case 'voice_note':
        return Icons.mic_rounded;
      case 'audio_file':
        return Icons.audiotrack_rounded;
      case 'video_file':
        return Icons.videocam_rounded;
      case 'lesson_plan':
        return Icons.description_rounded;
      case 'teaching_material':
        return Icons.menu_book_rounded;
      default:
        return Icons.attach_file_rounded;
    }
  }

  Color _color(String type) {
    switch (type) {
      case 'video_link':
        return const Color(0xFF6366F1);
      case 'voice_note':
        return const Color(0xFFEC4899);
      case 'audio_file':
        return const Color(0xFF8B5CF6);
      case 'video_file':
        return const Color(0xFFEF4444);
      case 'lesson_plan':
        return const Color(0xFF059669);
      case 'teaching_material':
        return const Color(0xFFF59E0B);
      default:
        return kPrimaryBlue;
    }
  }

  @override
  Widget build(BuildContext context) {
    final taskList = AppState.getTasksFor(date, periodIdx);
    return Column(
      children: [
        Expanded(
          child: taskList.isEmpty
              ? Center(
                  child: Column(
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: [
                      Icon(
                        Icons.attach_file_rounded,
                        size: 48,
                        color: kBorderCol,
                      ),
                      const SizedBox(height: 12),
                      const Text(
                        'No tasks attached yet',
                        style: TextStyle(
                          color: kMidText,
                          fontSize: 14,
                          fontWeight: FontWeight.w600,
                        ),
                      ),
                      const SizedBox(height: 4),
                      const Text(
                        'Tap + to add a task',
                        style: TextStyle(color: kMidText, fontSize: 12),
                      ),
                    ],
                  ),
                )
              : ListView.builder(
                  padding: const EdgeInsets.fromLTRB(16, 8, 16, 8),
                  itemCount: taskList.length,
                  itemBuilder: (_, i) {
                    final t = taskList[i];
                    final color = _color(t['type'] ?? '');
                    return Container(
                      margin: const EdgeInsets.only(bottom: 10),
                      decoration: BoxDecoration(
                        color: color.withValues(alpha: 0.06),
                        borderRadius: BorderRadius.circular(14),
                        border: Border.all(color: color.withValues(alpha: 0.2)),
                      ),
                      child: ListTile(

  bool get _isText => _type == 'lesson_plan' || _type == 'teaching_material';
  bool get _isFile => !_isUrl && !_isText;

    switch (_type) {
      case 'audio_file':
        return FileType.audio;
      case 'video_file':
        return FileType.video;
      case 'voice_note':
        return FileType.audio;
      default:
        return FileType.any;
    }

    setState(() => _picking = true);

          _pickedFileName = f.name;
          _pickedFilePath = f.path ?? f.name;

class TimetableTimeline extends StatelessWidget {

    super.key,

  @override
  Widget build(BuildContext context) => LayoutBuilder(
    builder: (_, c) {

                borderRadius: BorderRadius.circular(1),
              ),
            ),
          ),
          // Ticks
          ...List.generate(
            totalPeriods,
            (i) => Positioned(
              left: 20,
              top: i * rowH + rowH * 0.5 - 1,

                width: 14,
                height: 2,

                  color: kBorderCol,
                  borderRadius: BorderRadius.circular(1),

            child: Row(
              mainAxisSize: MainAxisSize.min,

          Positioned(

class _Arrow extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    canvas.drawPath(
      Path()
        ..moveTo(size.width, 0)
        ..lineTo(0, size.height / 2)
        ..lineTo(size.width, size.height)
        ..close(),
      Paint()..color = const Color(0xFFEF4444),
    );
  }

  @override
  bool shouldRepaint(_Arrow _) => false;
}

}

class _SchedulePageState extends State<SchedulePage>
    with SingleTickerProviderStateMixin {
  late TabController _tabs;
  late DateTime _today;
  late DateTime _weekStart;
  static const _dayNames = [
    'Monday',
    'Tuesday',
    'Wednesday',
    'Thursday',
    'Friday',
  ];

  @override
  void initState() {
    super.initState();
    _today = DateTime.now();
    _weekStart = _today.subtract(Duration(days: _today.weekday - 1));
    final todayTab = (_today.weekday - 1).clamp(0, 4);
    _tabs = TabController(length: 5, vsync: this, initialIndex: todayTab);
  }

  @override
  void dispose() {
    _tabs.dispose();
    super.dispose();
  }

  DateTime _dateForTab(int tab) => _weekStart.add(Duration(days: tab));
  String _dk(DateTime d) => AppState.dateKey(d);
  bool _isToday(int tab) {
    final d = _dateForTab(tab);
    return d.year == _today.year &&
        d.month == _today.month &&
        d.day == _today.day;
  }

  void _toggleRelief(int tab, int idx) {
    final k = _dk(_dateForTab(tab));
    setState(() {
      AppState.reliefMap.putIfAbsent(k, () => {});
      if (AppState.reliefMap[k]!.contains(idx)) {
        AppState.reliefMap[k]!.remove(idx);
      } else {
        AppState.reliefMap[k]!.add(idx);
      }
    });
  }

  @override
  Widget build(BuildContext context) => Scaffold(
    backgroundColor: kBgLight,
    body: SafeArea(
      child: Column(
        children: [
          _buildHeader(),
          const SizedBox(height: 8),
          _buildWeekNav(),
          const SizedBox(height: 6),
          _buildTabs(),
          const SizedBox(height: 6),

            child: TabBarView(
              controller: _tabs,

  );

  Widget _buildDayView(int tab) {
    final date = _dateForTab(tab);
    final dk = _dk(date);
    final relieved = AppState.reliefMap[dk] ?? {};
    final isToday = _isToday(tab);
    final n = AppState.periods > 0 ? AppState.periods : 8;
    final start = AppState.firstStart ?? const TimeOfDay(hour: 7, minute: 30);
    final dur = AppState.duration > 0 ? AppState.duration : 40;
    final intS = AppState.intervalStart;
    final intE = AppState.intervalEnd;
    final saved = AppState.timetable.length > tab
        ? AppState.timetable[tab]
        : <Map<String, String>>[];

    final List<Map<String, dynamic>> sched = [];
    int cur = _toMin(start);
    for (int i = 0; i < n; i++) {
      if (intS != null && intE != null && cur == _toMin(intS)) {
        cur = _toMin(intE);
      }
      final ps = TimeOfDay(hour: cur ~/ 60, minute: cur % 60);
      final end = cur + dur;
      final pe = TimeOfDay(hour: end ~/ 60, minute: end % 60);
      final m = i < saved.length ? saved[i] : <String, String>{};
      sched.add({
        'index': i,
        'number': i + 1,
        'subject': m['subject'] ?? '',
        'class': m['class'] ?? '',
        'isOff': (m['subject'] ?? '').isEmpty || (m['type'] ?? '') == 'off',
        'isRelief': relieved.contains(i),
        'startTime': ps,
        'endTime': pe,
      });
      cur = end;
    }

    return Column(
      children: [
        Padding(
          padding: const EdgeInsets.fromLTRB(16, 8, 16, 8),
          child: Row(
            children: [
              Container(
                padding: const EdgeInsets.symmetric(
                  horizontal: 10,
                  vertical: 5,
                ),
                decoration: BoxDecoration(
                  color: isToday ? kPrimaryBlue : kBgLight,
                  borderRadius: BorderRadius.circular(20),
                  border: Border.all(
                    color: isToday ? kPrimaryBlue : kBorderCol,
                  ),
                ),
                child: Text(
                  '${_dayNames[tab]}, ${date.day} ${['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'][date.month - 1]}',
                  style: TextStyle(
                    color: isToday ? Colors.white : kMidText,
                    fontSize: 12,
                    fontWeight: FontWeight.w700,
                  ),
                ),
              ),
              const Spacer(),
              if (relieved.isNotEmpty) ...[
                Container(
                  padding: const EdgeInsets.symmetric(
                    horizontal: 10,
                    vertical: 5,
                  ),
                  decoration: BoxDecoration(
                    color: const Color(0xFFFEF9C3),
                    borderRadius: BorderRadius.circular(20),
                  ),
                  child: Text(
                    '${relieved.length} relief',
                    style: const TextStyle(
                      color: Color(0xFF78350F),
                      fontSize: 12,
                      fontWeight: FontWeight.w700,
                    ),
                  ),
                ),
                const SizedBox(width: 8),
                GestureDetector(
                  onTap: () => setState(() => AppState.reliefMap.remove(dk)),
                  child: Container(
                    padding: const EdgeInsets.symmetric(
                      horizontal: 10,
                      vertical: 5,
                    ),
                    decoration: BoxDecoration(
                      color: kBgLight,
                      borderRadius: BorderRadius.circular(20),
                      border: Border.all(color: kBorderCol),
                    ),
                    child: const Text(
                      'Clear all',
                      style: TextStyle(
                        color: kMidText,
                        fontSize: 12,
                        fontWeight: FontWeight.w600,
                      ),
                    ),
                  ),
                ),
              ],
            ],
          ),
        ),
        Expanded(
          child: ListView.builder(
            padding: const EdgeInsets.fromLTRB(16, 0, 16, 16),
            itemCount: sched.length,
            itemBuilder: (_, i) {
              final s = sched[i];
              final isOff = s['isOff'] as bool;
              final isRelief = s['isRelief'] as bool;
              Color bg, border, tc;
              if (isOff) {
                bg = const Color(0xFFE8E8EA);
                border = const Color(0xFFCBCBD0);
                tc = const Color(0xFF5C5C6E);
              } else if (isRelief) {
                bg = const Color(0xFFFEF9C3);
                border = const Color(0xFFFDE68A);
                tc = const Color(0xFF78350F);
              } else if ((s['subject'] as String).isNotEmpty &&
                  (s['class'] as String).isNotEmpty) {
                bg = const Color(0xFFD1FAE5);
                border = const Color(0xFF6EE7B7);
                tc = const Color(0xFF065F46);
              } else {
                bg = Colors.white;
                border = kBorderCol;
                tc = kDarkText;
              }
              return Container(
                margin: const EdgeInsets.only(bottom: 10),
                decoration: BoxDecoration(
                  color: bg,
                  borderRadius: BorderRadius.circular(14),
                  border: Border.all(color: border, width: 1.5),
                  boxShadow: [
                    BoxShadow(
                      color: Colors.black.withValues(alpha: 0.05),
                      blurRadius: 3,
                      offset: const Offset(0, 1),
                    ),
                  ],
                ),
                child: Padding(
                  padding: const EdgeInsets.symmetric(
                    horizontal: 14,
                    vertical: 12,
                  ),
                  child: Row(
                    children: [
                      Container(
                        width: 32,
                        height: 32,
                        decoration: BoxDecoration(
                          shape: BoxShape.circle,
                          color: Colors.black.withValues(alpha: 0.1),
                        ),
                        child: Center(
                          child: Text(
                            '${s['number']}',
                            style: TextStyle(
                              color: tc,
                              fontSize: 13,
                              fontWeight: FontWeight.w800,
                            ),
                          ),
                        ),
                      ),
                      const SizedBox(width: 12),
                      Expanded(
                        child: Column(
                          crossAxisAlignment: CrossAxisAlignment.start,
                          children: [
                            Text(
                              isOff
                                  ? 'Off Period'
                                  : isRelief
                                  ? 'Relief'
                                  : '${s['subject']}',
                              style: TextStyle(
                                color: tc,
                                fontSize: 14,
                                fontWeight: FontWeight.w700,
                              ),
                            ),
                            Text(
                              '${_fmt12Tod(s['startTime'] as TimeOfDay)} – ${_fmt12Tod(s['endTime'] as TimeOfDay)}',
                              style: TextStyle(
                                color: tc.withValues(alpha: 0.65),
                                fontSize: 12,
                                fontWeight: FontWeight.w600,

                            ),
                          ],
                        ),
                      ),
                      if (!isOff &&
                          (s['class'] as String).isNotEmpty &&
                          !isRelief)
                        Container(
                          margin: const EdgeInsets.only(right: 10),
                          padding: const EdgeInsets.symmetric(
                            horizontal: 8,
                            vertical: 4,
                          ),
                          decoration: BoxDecoration(
                            color: const Color(0xFF34D399),
                            borderRadius: BorderRadius.circular(8),
                          ),
                          child: Text(
                            s['class'] as String,
                            style: const TextStyle(
                              color: Colors.white,
                              fontSize: 12,
                              fontWeight: FontWeight.w800,
                            ),
                          ),
                        ),
                      if (!isOff)
                        GestureDetector(
                          onTap: () => _toggleRelief(tab, s['index'] as int),
                          child: AnimatedContainer(
                            duration: const Duration(milliseconds: 200),
                            padding: const EdgeInsets.symmetric(
                              horizontal: 12,
                              vertical: 7,
                            ),
                            decoration: BoxDecoration(
                              color: isRelief
                                  ? const Color(0xFFFBBF24)
                                  : Colors.white.withValues(alpha: 0.6),
                              borderRadius: BorderRadius.circular(20),
                              border: Border.all(
                                color: isRelief
                                    ? const Color(0xFFFBBF24)
                                    : Colors.black.withValues(alpha: 0.15),
                              ),
                            ),
                            child: Row(
                              mainAxisSize: MainAxisSize.min,
                              children: [
                                Icon(
                                  isRelief
                                      ? Icons.swap_horiz_rounded
                                      : Icons.swap_horiz_outlined,
                                  color: isRelief

                                  isRelief ? 'Relief ✓' : 'Set Relief',

  String _gender = '';
  TimeOfDay? _fs, _le, _is, _ie;
  late double _fontSize;

  @override
  void initState() {
    super.initState();
    _nameCtrl = TextEditingController(text: AppState.name);
    _periodsCtrl = TextEditingController(text: '${AppState.periods}');
    _durationCtrl = TextEditingController(text: '${AppState.duration}');
    _gender = AppState.gender;
    _fs = AppState.firstStart;

    _is = AppState.intervalStart;
    _ie = AppState.intervalEnd;
    _fontSize = AppState.fontSizeNotifier.value;

  @override
  void dispose() {
    _nameCtrl.dispose();
    _periodsCtrl.dispose();
    _durationCtrl.dispose();
    super.dispose();
  }

  void _save() {
    AppState.name = _nameCtrl.text.trim();
    AppState.gender = _gender;
    AppState.periods = int.tryParse(_periodsCtrl.text) ?? AppState.periods;
    AppState.duration = int.tryParse(_durationCtrl.text) ?? AppState.duration;
    AppState.firstStart = _fs;
    AppState.lastEnd = _le;
    AppState.intervalStart = _is;
    AppState.intervalEnd = _ie;
    AppState.fontSizeNotifier.value = _fontSize;
    Navigator.of(context).pop();
  }
